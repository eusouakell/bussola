/* app.css */
@import "tailwindcss";

@custom-variant dark (&:where(.dark, .dark *));
@custom-variant reduced-transparency
  (@media (prefers-reduced-transparency: reduce));
@custom-variant no-glass
  (@supports not (backdrop-filter: blur(1px)));

@theme {
  --font-sans: "Nunito Sans", ui-sans-serif, system-ui;
  --font-mono: "JetBrains Mono", ui-monospace;

  --color-accent: #EC7000;
  --color-accent-cta: #DB6600;
  --color-accent-ink: #A84F00;
  --color-ink: #1E1A16;
  --color-ink-2: #4A423A;
  --color-ink-3: #675D53;
  --color-sand-50: #FBF7F2;
  --color-smoke-900: #1D1713;
  --color-ok: #1E7A45;
  --color-warn: #8A5200;
  --color-warn-fill: #F2A93B;
  --color-err: #B42318;

  --color-tag-diagnostico: #0B6B70;
  --color-tag-simulacao: #6A3FB5;
  --color-tag-recomendacao: #A84F00;
  --color-tag-acao: #1E7A45;

  --color-tint-objetivo: #EEDCC8;
  --color-tint-entender: #E4DCD2;
  --color-tint-antecipar: #E3D5EC;
  --color-tint-orientar: #F8BD88;
  --color-tint-agir: #CFE7D3;
  --color-tint-acompanhar: #F4CB7A;

  --radius-card: 22px;
  --radius-consent: 24px;
  --radius-field: 16px;
  --radius-control: 12px;

  --blur-glass-chrome: 24px;
  --blur-glass-card: 20px;
  --blur-glass-consent: 22px;

  --shadow-glass: inset 0 1px 0 rgb(255 255 255 / .9),
    0 22px 48px -24px rgb(150 78 20 / .22);
  --shadow-consent: 0 0 0 6px rgb(46 160 90 / .12),
    0 0 44px -6px rgb(46 160 90 / .30);

  --animate-drift: drift 26s ease-in-out infinite alternate;
  --animate-enter: enter .55s cubic-bezier(.2,.8,.2,1) both;
}
@utility glass-chrome {
  background: rgb(255 255 255 / .42);
  backdrop-filter: blur(var(--blur-glass-chrome)) saturate(150%);
  border: 1px solid rgb(255 255 255 / .62);
  box-shadow: inset 0 1px 0 rgb(255 255 255 / .9),
    0 10px 30px -14px rgb(150 78 20 / .22);
  @variant dark { background: rgb(14 11 9 / .36);
    border-color: rgb(255 255 255 / .10); }
}
@utility glass-card {
  background: rgb(255 252 248 / .80);
  backdrop-filter: blur(var(--blur-glass-card)) saturate(140%);
  border: 1px solid rgb(255 255 255 / .78);
  border-radius: var(--radius-card);
  box-shadow: var(--shadow-glass);
  @variant dark { background: rgb(24 19 15 / .80);
    border-color: rgb(255 255 255 / .10); }
}
@utility glass-consent {
  background: rgb(255 255 255 / .88);
  backdrop-filter: blur(var(--blur-glass-consent)) saturate(150%);
  border: 1.5px solid rgb(30 122 69 / .55);
  border-radius: var(--radius-consent);
  box-shadow: var(--shadow-glass), var(--shadow-consent);
}
/* fallback sólido para os 3 níveis */
@layer utilities {
  .glass-chrome, .glass-card, .glass-consent {
    @variant reduced-transparency {
      backdrop-filter: none; background: var(--color-sand-50); }
    @variant no-glass { background: var(--color-sand-50); }
  }
}
/* ambiente dinâmico: CSS puro, sem canvas/WebGL */
@property --amb-1 { syntax: "<color>"; inherits: true;
  initial-value: #EEDCC8; }
[data-estado="orientar"] { --amb-1: var(--color-tint-orientar); }
.ambient { transition: --amb-1 600ms, --amb-2 600ms, --amb-3 600ms; }
.blob { animation: var(--animate-drift); }
